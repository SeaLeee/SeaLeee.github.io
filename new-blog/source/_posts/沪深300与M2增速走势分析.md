# 投资信号生成逻辑
def generate_signal(hs300_price, m2_growth, correlation):
    signal_score = 0
    
    # M2增速信号
    if m2_growth > 8.0:
        signal_score += 2  # 强烈利好
    elif m2_growth > 7.5:
        signal_score += 1  # 适度利好
    elif m2_growth < 6.5:
        signal_score -= 2  # 利空
    
    # 相关性背离信号
    if correlation < 0.4:
        signal_score += 1  # 可能反转
    
    # 技术面信号
    ma30 = calculate_ma(hs300_price, 30)
    if hs300_price > ma30 * 1.02:
        signal_score += 1
    elif hs300_price < ma30 * 0.98:
        signal_score -= 1
    
    return signal_score

# 当前信号
current_signal = generate_signal(3850.23, 7.8, 0.65)
print(f"投资信号评分: {current_signal}/5")
